import time
from playwright.sync_api import sync_playwright

BASE_URL = "https://motionsites.ai"

def scrape_motionsites():
    with sync_playwright() as p:
        # Launch browser with clipboard permissions enabled
        browser = p.chromium.launch(headless=False)
        context = browser.new_context(permissions=["clipboard-read", "clipboard-write"])
        page = context.new_page()
        
        print("Loading MotionSites catalog...")
        page.goto(BASE_URL)
        page.wait_for_load_state("networkidle")
        
        # Step 1: Scroll to dynamically load all free items
        last_height = page.evaluate("document.body.scrollHeight")
        while True:
            page.evaluate("window.scrollTo(0, document.body.scrollHeight);")
            time.sleep(2)  # Wait for pagination/lazy load
            new_height = page.evaluate("document.body.scrollHeight")
            if new_height == last_height:
                break
            last_height = new_height
            
        # Extract all template page links
        # Adjust selector based on actual class names of the card anchors
        links = page.locator("a[href*='/design/'], a[href*='/template/']").all_inner_texts()
        urls = [page.locator(f"text={link}").get_attribute("href") for link in links if link]
        # Alternative: grab href directly via JS evaluation
        urls = page.evaluate('''() => {
            return Array.from(document.querySelectorAll('a'))
                .map(a => a.href)
                .filter(href => href.includes('/app') || href.includes('/design')); 
        }''')
        
        unique_urls = list(set(urls))
        print(f"Found {len(unique_urls)} potential template links.")
        
        scraped_prompts = []
        
        # Step 2: Extract text from each template page
        for index, url in enumerate(unique_urls):
            try:
                print(f"Scraping [{index+1}/{len(unique_urls)}]: {url}")
                page.goto(url)
                page.wait_for_load_state("domcontentloaded")
                
                # Fetch metadata
                title = page.locator("h1").inner_text() if page.locator("h1").count() > 0 else "Untitled"
                
                # Target and click the "Copy Full Prompt" button
                copy_button = page.locator("button:has-text('Copy Full Prompt'), button:has-text('Copy Prompt')")
                
                if copy_button.count() > 0:
                    copy_button.click()
                    time.sleep(0.5) # Wait for clipboard write action
                    
                    # Read the prompt string out of the browser's clipboard context
                    prompt_text = page.evaluate("navigator.clipboard.readText()")
                    
                    scraped_prompts.append({
                        "title": title,
                        "url": url,
                        "prompt": prompt_text
                    })
                else:
                    print(f"Skipping premium or uncopyable item at: {url}")
                    
            except Exception as e:
                print(f"Failed to scrape {url}: {e}")
                continue
                
        # Save dataset to a local structured JSON file
        with open("motionsites_free_prompts.json", "w", encoding="utf-8") as f:
            json.dump(scraped_prompts, f, indent=4, ensure_ascii=False)
            
        print(f"Successfully exported {len(scraped_prompts)} prompts to motionsites_free_prompts.json!")
        browser.close()

if __name__ == "__main__":
    scrape_motionsites()

def scrape_all_prompts():
    print("🚀 Initializing Scraping Sequence...")
    
    with sync_playwright() as p:
        # Launching with headless=False keeps browser focused for clipboard permission access
        browser = p.chromium.launch(headless=False)
        context = browser.new_context(
            permissions=["clipboard-read", "clipboard-write"],
            viewport={"width": 1280, "height": 800}
        )
        page = context.new_page()
        
        print(f"📡 Navigating to Template Catalog: {BASE_URL}")
        page.goto(BASE_URL)
        page.wait_for_load_state("networkidle")
        
        # Phase 1: Infinite scroll script to trigger lazy-loaded cards
        print("🔄 Scrolling page to discover all elements...")
        last_height = page.evaluate("document.body.scrollHeight")
        while True:
            page.evaluate("window.scrollTo(0, document.body.scrollHeight);")
            time.sleep(2.5)  # Safe buffer interval
            new_height = page.evaluate("document.body.scrollHeight")
            if new_height == last_height:
                break
            last_height = new_height

        # Phase 2: Grab all individual template page links via custom evaluation
        urls = page.evaluate('''() => {
            return Array.from(document.querySelectorAll('a'))
                .map(a => a.href)
                .filter(href => href.includes('/design/') || href.includes('/template/') || href.includes('/app/'));
        }''')
        
        # Deduplicate endpoints
        unique_urls = sorted(list(set(urls)))
        print(f"🎯 Discovered {len(unique_urls)} distinct template paths to parse.")
        
        scraped_data = []

        # Phase 3: Crawl loop over discovered items
        for index, url in enumerate(unique_urls):
            try:
                print(f"📖 [{index + 1}/{len(unique_urls)}] Processing: {url}")
                page.goto(url)
                page.wait_for_load_state("domcontentloaded")
                time.sleep(1) # Prevent aggressive server pinging
                
                title = page.locator("h1").inner_text() if page.locator("h1").count() > 0 else "Unnamed Template"
                
                # Use semantic text selector for button interaction
                copy_btn = page.locator("button:has-text('Copy Full Prompt'), button:has-text('Copy Prompt')")
                
                if copy_btn.count() > 0:
                    # Bring window to foreground to fulfill strict browser security constraints on clipboard APIs
                    page.bring_to_front()
                    copy_btn.click()
                    time.sleep(0.5)  # Await data transit to virtual clipboard
                    
                    # Read values directly from page instance browser shell
                    prompt_text = page.evaluate("navigator.clipboard.readText()")
                    
                    if prompt_text:
                        scraped_data.append({
                            "title": title,
                            "url": url,
                            "prompt": prompt_text.strip()
                        })
                        print(f"✅ Success: Extracted {len(prompt_text)} characters.")
                    else:
                        print("⚠️ Empty clipboard content.")
                else:
                    print("🔒 Locked or Premium tier layout - Skipping interaction.")
                    
            except Exception as error:
                print(f"❌ Execution Failure at {url}: {error}")
                continue

        # Phase 4: Persistence formatting
        output_file = "motionsites_prompts_archive.json"
        with open(output_file, "w", encoding="utf-8") as f:
            json.dump(scraped_data, f, indent=4, ensure_ascii=False)
            
        print(f"\n🎉 Task Completed! {len(scraped_data)} prompts archived inside '{output_file}'!")
        browser.close()

if __name__ == "__main__":

        # Launching localized Chromium bundle
        # headless=False ensures Windows updates the active OS clipboard memory buffer
        browser = p.chromium.launch(headless=False)
        
        context = browser.new_context(
            permissions=["clipboard-read", "clipboard-write"],
            viewport={"width": 1366, "height": 768}
        )

        page.goto(BASE_URL)
        page.wait_for_load_state("networkidle")
